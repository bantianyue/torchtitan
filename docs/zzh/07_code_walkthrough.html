2. parallelize_llama（并行编排）

models/llama3/parallelize.py 是「把各种 PT-D 并行与训练技巧应用到模型」的总指挥。注意顺序：CP/TP 包裹 → async TP → 激活重算 → compile → FSDP。

def parallelize_llama(model, *, parallel_dims, training, parallelism,
                      compile_config, ac_config, dump_folder):
    if spmd_backend in ("full_dtensor", "spmd_types"):
        validate_config(parallel_dims, model)
        model.parallelize(parallel_dims)                  # 声明式切分落地
    else:
        if parallel_dims.cp_enabled:
            apply_cp_to_forward([b.attention.inner_attention for b in model.layers.values()],
                                  parallel_dims.get_mesh("cp"))
        if parallel_dims.tp_enabled:
            model.parallelize(parallel_dims)              # TP/SP 声明落地

    if parallel_dims.tp_enabled:
        maybe_enable_async_tp(parallelism, compile_config, parallel_dims.get_mesh("tp"))

    if ac_config is not None:
        ac_config.build(dump_folder=dump_folder).apply(model)   # 激活重算（在 compile 前）

    model_compile_enabled = compile_config.enable and "model" in compile_config.components
    if model_compile_enabled:
        apply_compile(model, compile_config)            # compile（在 FSDP 前）

    # FSDP/HSDP：始终调用；degree=1 时仅是 no-op 的 all-gather，但装好 MixedPrecisionPolicy
    dp_mesh, dp_mesh_dims = resolve_fsdp_mesh(parallel_dims)
    apply_fsdp_to_decoder(model, dp_mesh,
        param_dtype=TORCH_DTYPE_MAP[training.mixed_precision_param],
        reduce_dtype=TORCH_DTYPE_MAP[training.mixed_precision_reduce],
        pp_enabled=parallel_dims.pp_enabled, cpu_offload=training.enable_cpu_offload,
        reshard_after_forward_policy=parallelism.fsdp_reshard_after_forward,
        dp_mesh_dims=dp_mesh_dims, enable_symm_mem=parallelism.enable_fsdp_symm_mem)
    return model
顺序为什么是这个顺序
AC 包裹要在 compile 之前（让编译器看见 wrapped 图）；compile 要在 FSDP 之前（FSDP 的 composable 钩子应与编译后的图配合）；FSDP 始终最后，因为它要装 MixedPrecisionPolicy 并接管参数分片。CP/TP 的 parallelize() 在 AC/compile 之前。

3. apply_fsdp_to_decoder（FSDP 落地）

选取稠密/MoE 共用的 FSDP 应用函数。这里截取最关键的分片逻辑：

def apply_fsdp_to_decoder(model, dp_mesh, param_dtype, reduce_dtype, pp_enabled, ...):
    mp_policy = MixedPrecisionPolicy(param_dtype=param_dtype, reduce_dtype=reduce_dtype,
                                  cast_forward_inputs=False)
    fsdp_config = {"mesh": dp_mesh, "mp_policy": mp_policy}
    if dp_mesh_dims is not None: fsdp_config["dp_mesh_dims"] = dp_mesh_dims
    if cpu_offload: fsdp_config["offload_policy"] = CPUOffloadPolicy()

    reshard = get_fsdp_reshard_after_forward_policy(reshard_after_forward_policy, pp_enabled)

    # 权重共享：把 tok_emb/norm/lm_head 合成一个 FSDP 单元
    if model.enable_weight_tying:
        fully_shard([m for m in (model.tok_embeddings, model.norm, model.lm_head) if m],
                     **fsdp_config, reshard_after_forward=(reshard=="always"))
    else:
        fully_shard(model.tok_embeddings, **fsdp_config, reshard_after_forward=reshard)
        fully_shard([model.norm, model.lm_head], **fsdp_config,
                     reshard_after_forward=(reshard=="always"))   # 末层不 reshard 以省 prefetch

    # 逐 transformer block 分片（MoE 用 shard_placement_fn 区分专家/非专家）
    for layer_id, block in model.layers.items():
        if getattr(block, "moe_enabled", False):
            # ep>1: 专家走 edp_mesh，其余走 dp_mesh；ep=1 且专家数
            ...
        else:
            fully_shard(block, **fsdp_config, reshard_after_forward=reshard)

    fully_shard(model, **fsdp_config)
    disable_fsdp_gradient_division(model)   # 关自动除法，改由 global token 数归一化
    if "dp_replicate" in (dp_mesh.mesh_dim_names or ()): logger.info("Applied HSDP")
    else: logger.info("Applied FSDP")
EP 下为何要显式预取
当 ep_degree > 1，MoE 的 D2H 同步会干扰 FSDP 隐式预取。函数末尾因此显式 set_modules_to_forward_prefetch / set_modules_to_backward_prefetch，把相邻 block 串成预取链（tok_emb → block0 → … → norm/lm_head，反向反之）。

4. pipeline_llm（PP 拆分与调度）

def pipeline_llm(model, *, parallel_dims, training, parallelism, compile_config,
                  ac_config, dump_folder, device, model_config, parallelize_fn, loss_fn):
    pp_mesh = parallel_dims.get_mesh("pp")
    (num_virtual_stages, num_layers, in_w, out_w) = _get_pipeline_metadata(...)
    module_names_per_stage = parallelism.module_fqns_per_model_part or \
        _generate_llm_fqn_per_model_part(num_virtual_stages, num_layers, in_w, out_w)

    get_mesh_cb = _build_get_mesh_callback(parallel_dims)   # PP 跨 stage 重建 DTensor
    stages, model_parts = _pipeline_module_split(model, pp_mesh, schedule, device,
                                              module_names_per_stage, get_mesh=get_mesh_cb)

    # 每个 model_part 单独应用 SPMD/compile/optimizer
    for i, m in enumerate(model_parts):
        m = parallelize_fn(m, ...)
        model_parts[i] = m
        stages[i].submod = m

    pp_schedule = _build_pipeline_schedule(parallelism=parallelism,
        local_batch_size=training.local_batch_size, stages=stages, loss_fn=loss_fn)

    has_first_stage = any(s.is_first for s in stages)
    has_last_stage  = any(s.is_last  for s in stages)
    return pp_schedule, model_parts, has_first_stage, has_last_stage

关键细节：PP 跨 stage 传递的是普通张量（ProcessGroup 不可序列化），所以每个 stage 用 get_mesh_cb 回调重新拿到本地 DeviceMesh，把收到的张量重新包成正确 placement 的 DTensor。

5. forward_backward_step（前向反向）

def forward_backward_step(self, *, input_dict, labels, global_valid_tokens):
    inputs, labels, extra_kwargs = self.post_dataloading_process(input_dict, labels)

    if parallel_dims.pp_enabled:
        with self.train_context():
            targets, losses = (labels, []) if self.pp_has_last_stage else (None, None)
            if self.pp_has_first_stage:
                self.pp_schedule.step(inputs, **extra_kwargs, target=targets,
                                      losses=losses, loss_kwargs={"global_valid_tokens": ...})
            if self.pp_has_last_stage:
                loss = torch.sum(torch.stack(losses)).to(self.device)
            else:
                loss = torch.tensor([-1.0], device=self.device)
    else:
        with self.train_context():
            pred = model_parts[0](inputs, **extra_kwargs)
            loss, _ = self.loss_fn(pred, labels, global_valid_tokens)
            with spmd.no_typecheck():      # 反向不需类型传播
                loss.backward()
    return loss
PP 下非末级 rank 的 loss 是占位
只有末级 stage 持有 labels 与 targets，能算出真实 loss；其他 rank 返回 -1.0 占位，仅在末级做有意义的聚合。这也是为什么 train_step 里 loss 聚合要区分 DP/CP 是否开启。

6. ChunkedLossWrapper 关键段

# 切 N 段 → 逐 chunk lm_head + CE + backward → 汇回 Decoder
h_chunks = [c.detach().requires_grad_(requires_grad) for c in _chunk(hidden_states)]
grad_accumulator = GradAccumulator(hidden_states, num_chunks=num_chunks, dtype=torch.float32)

if fsdp_enabled:                       # 让权重在整个 chunk 循环保持 unsharded
    lm_head.set_reshard_after_forward(False)
    lm_head.set_reshard_after_backward(False)
    lm_head.set_requires_gradient_sync(False, recurse=False)

for i, (h_chunk, label_chunk) in enumerate(zip(h_chunks, label_chunks)):
    if fsdp_enabled and i == last_idx:
        lm_head.set_requires_gradient_sync(True, recurse=False)  # 最后一 chunk 才触发 reduce-scatter
    logits = lm_head(h_chunk)
    chunk_loss, _ = self.loss_fn(logits, label_chunk, global_valid_tokens, **chunk_inputs)
    total_loss = total_loss + chunk_loss.detach()
    if requires_grad:
        chunk_loss.backward()
        grad_accumulator.add(h_chunk.grad); h_chunk.grad = None

accumulated_grad = grad_accumulator.result().to(hidden_states.dtype)
loss = self._gradient_backprop(hidden_states, accumulated_grad, total_loss, lm_head, fsdp_enabled)
